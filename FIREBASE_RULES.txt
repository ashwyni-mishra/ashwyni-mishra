# FIREBASE SECURITY RULES
# Copy these into your Firebase Console to fix "Missing or insufficient permissions"

# --- 1. FIRESTORE RULES ---
# Path: Firestore Database > Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Public read access for portfolio data
    match /portfolio/data {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Private access for user-specific settings (MFA, etc.)
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Messages from the contact form
    match /messages/{messageId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

    // Newsletter subscribers
    match /subscribers/{subscriberId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }
  }
}


# --- 2. STORAGE RULES ---
# Path: Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /uploads/{allPaths=**} {
      // Allow anyone to view uploaded assets (images/resumes)
      allow read: if true;
      // Only authenticated admins can upload or delete
      allow write: if request.auth != null;
    }
  }
}
